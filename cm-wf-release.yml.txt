name: Release APK

# 安卓统一发布入口：一处触发，出签名的 APK。
#
# 触发方式：
#   - 推送 v* 标签（例如 v4.0.5）→ 自动发布
#   - Actions 页面手动 Run workflow → 用于补发/重出
#
# 实际构建走「可复用工作流」build-apk.yml（内容原样保留）。
# 原生鸿蒙已归档（docs/archive/harmony-archive/），发布流程不再包含鸿蒙任务。
on:
  workflow_dispatch:
  push:
    tags:
      - 'v*'

permissions:
  contents: write

jobs:
  android:
    name: Android APK
    uses: ./.github/workflows/build-apk.yml
    with:
      # v* 标签发布 = 调试版（默认自动上调试版蒲公英 + 产物仓库）
      # 正式版要单独手动 Run build-apk.yml 并把通道选成 release。
      channel: debug
      upload_pgyer: true
    secrets: inherit
