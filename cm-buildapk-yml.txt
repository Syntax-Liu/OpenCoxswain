name: Build Release APK

on:
  # 手动触发：Actions 页面点 Run workflow（只出这一端，用于单独出包 / 发正式版）
  workflow_dispatch:
    inputs:
      channel:
        description: '分发通道: debug=调试版蒲公英 / release=正式版蒲公英'
        required: false
        type: choice
        options:
          - debug
          - release
        default: debug
  # tag v* 的统一入口在 release.yml，由它编排调用本工作流；这里声明为可复用工作流
  workflow_call:
    inputs:
      # 分发通道：debug（默认，v* 发布自动进调试版蒲公英）/ release（正式版，仅手动 Run 触发）。
      # 两个通道各对应蒲公英一个独立应用（一把 key 一条固定链接），调试/正式互不覆盖。
      channel:
        required: false
        type: string
        default: debug
      # 是否上传蒲公英。false = 只进产物仓库（Android-Latest），测试用户从 GitHub 自行取包。
      upload_pgyer:
        required: false
        type: boolean
        default: true
      # CloudBase 部署密钥：release 通道上传自托管直链时需要。调用方 release.yml 用
      # `secrets: inherit` 透传；手动 workflow_dispatch 直接读仓库 Secrets。
    secrets:
      CLOUDBASE_SECRET_ID:
        required: false
      CLOUDBASE_SECRET_KEY:
        required: false

# 流水线分两段，**构建与发布是两个独立 job**：
#
#   build   → 自测 → 单测 → 出包 → 权限断言 → 把产物传到本次 run 的附件（actions/upload-artifact）
#   publish → 从附件取回同一份 APK → 产物仓库（Android-Latest）→ 自托管直链 + 四项校验 → 蒲公英
#
# 为什么拆开（2026-11-08 事故）：runner 到 CloudBase 的跨境网络抽风时，`tcb hosting deploy`
# 内部的串行重试会拖 36 分钟才失败；它原先和出包挤在同一个 job 里，于是「包其实已经构好了，
# 但整条发版被判红」，还得多花十分钟重建一遍。拆开后看两个 job 的颜色就够了：
#   build 绿 + Publish 绿 = 发版完整成功；
#   build 绿 + Publish 红 = 安装包已产出可取，只是对外分发没完成。
# Actions 里点「Re-run failed jobs」只重跑 Publish，不会重新构建，也不浪费一次出包。
#
# 产物仓库为什么用发布而不是 Actions 附件：附件挂在每一次运行下面，几十次构建就是几十份包；
# 而 Android-Latest 发布每次覆盖掉上一版附件，永远只有一份最新安装包。
#
# 蒲公英那一步用仓库 Settings → Secrets 里的 PGYER_API_KEY；没配就自动跳过，不影响出包。
# 最小权限：工作流级只给读。真正要写仓库的只有 publish job（gh release 覆盖 Android-Latest），
# 它在 job 内单独声明 contents: write；build job 只做 checkout / 构建 / 传附件，读权限足够。
permissions:
  contents: read

# 发版串行：Android-Latest 是仓库里唯一的一个固定标签，所有发版路径（tag 触发的 release.yml、
# 两个编排工作流、以及直接 dispatch 本工作流）都落在同一个组，不按 ref 分组 ——
# 否则「tag 发版」与「main 上手动 dispatch」会各成一组、并发去删/传同一批附件。
# cancel-in-progress: false = 排队而不是取消旧的：publish 是「先删旧附件再传新包」，
# 中途取消会留下半套产物；排队则保证每次发布要么完整成功、要么完整失败。
concurrency:
  group: android-release
  cancel-in-progress: false

jobs:
  # ---------------------------------------------------------------------------- 出包
  build:
    runs-on: ubuntu-latest
    # 构建本身约 10~12 分钟，给三倍余量；发布的重试与超时在 Publish job 里单独兜底。
    timeout-minutes: 40

    outputs:
      version: ${{ steps.ver.outputs.name }}
      sha256: ${{ steps.package.outputs.sha256 }}
      size: ${{ steps.package.outputs.size }}

    env:
      # 签名口令不在仓库里（体检 E-01）：仓库 gradle.properties 只有占位，
      # 构建时从这里注入，Gradle 按 ORG_GRADLE_PROJECT_<名字> 自动当成同名属性读取。
      # 没配这两个 Secrets 时 app/build.gradle 会直接报错并提示怎么配，不会静默出一把无签名的包。
      ORG_GRADLE_PROJECT_coursemeowStorePassword: ${{ secrets.COURSEMEOW_STORE_PASSWORD }}
      ORG_GRADLE_PROJECT_coursemeowKeyPassword: ${{ secrets.COURSEMEOW_KEY_PASSWORD }}

    steps:
      - name: Checkout
        uses: actions/checkout@v7

      - name: Set up JDK 21
        uses: actions/setup-java@v6
        with:
          distribution: temurin
          java-version: '21'

      # 项目自带 Gradle wrapper（gradlew + gradle/wrapper/）。这里**不写死版本号**：
      # setup-gradle 会读 wrapper 里的版本来装，并把它一起做进缓存 ——
      # 写死的话会和 wrapper 各下一份 8.7，白等一次一百多兆。
      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v6

      # 构建机预装了 SDK，但 sdkmanager 不在 PATH 上（原来直接调用只会报
      # "command not found" 然后被 || true 吞掉）。这里走绝对路径，
      # 实在没有就退回预装 SDK，不阻断构建。
      - name: Ensure Android SDK components
        run: |
          SDK="${ANDROID_HOME:-${ANDROID_SDK_ROOT:-/usr/local/lib/android/sdk}}"
          SDKMGR="$SDK/cmdline-tools/latest/bin/sdkmanager"
          if [ -x "$SDKMGR" ]; then
            yes 2>/dev/null | "$SDKMGR" --licenses > /dev/null 2>&1 || true
            "$SDKMGR" "platforms;android-34" "build-tools;34.0.0" > /dev/null \
              || echo "::warning::sdkmanager 安装失败，改用构建机预装的 SDK"
          else
            echo "未找到 $SDKMGR，直接使用构建机预装的 SDK（android-34 / build-tools 34.0.0）"
          fi

      # local.properties 里的 sdk.dir 是本地 Windows 路径，在 CI 上必须去掉，
      # 让 AGP 走 ANDROID_HOME（GitHub 的 runner 已预装 SDK）
      - name: Drop local.properties
        run: rm -f local.properties

      # 仓库是在 Windows 上提交的，git 里 gradlew 可能没带可执行位；
      # Linux runner 上直接 ./gradlew 会 Permission denied，先补一下。
      - name: Make gradlew executable
        run: chmod +x ./gradlew

      # 下载页只有 metaUrl 非空时才请求元数据（避免线上还没有 download-meta.json 时刷 404）。
      # 正式通道在这里做硬检查：元数据传了但页面不读，或者页面读的地址不对，都会让下载页
      # 停在旧值 / 404。首次启用时把 site.js 的 metaUrl 设为下面这个值并部署站点即可，之后不用再动。
      - name: Verify download metadata enabled (release channel)
        if: ${{ inputs.channel == 'release' }}
        run: |
          node -e "const fs=require('fs'); const s=fs.readFileSync('web/assets/site.js','utf8'); const m=s.match(/metaUrl\s*:\s*'([^']*)'/); process.exit(m && m[1]==='/download/download-meta.json'?0:1)" \
            || { echo "::error::web/assets/site.js 的 metaUrl 不是 /download/download-meta.json：先把元数据开关打开并部署站点，再发正式版"; exit 1; }

      # 自测、文案体检、适配器体检与回放都只要几秒，放在编译前面：失败时能立刻看出是哪一类问题，
      # 不用等十分钟的构建跑完才知道。
      # 适配器这两步在 CI 上是硬门槛（本机没 node 时 Gradle 侧才会跳过）：
      # 少一步就等于「无回放用例也能标 A」，准入规则会被悄悄绕过。
      - name: Run self-tests
        run: |
          node tools/selftest/run-all.js
          node tools/copy-audit.js
          node tools/check-adapters.mjs
          node tools/replay-adapters.mjs

      # 本地单元测试：app/src/test 盯「客户端套餐与云端是否一致」「支付宝唤端链接的编码」，
      # 这两处错了不会报错，只会静默地收错钱 / 付不了款；core/src/test 是纯 JVM 的解析与
      # 时间计算用例。两条都要跑 —— 只跑 app 会让 core 的回归悄悄漏过去。
      - name: Run unit tests
        run: ./gradlew --no-daemon :app:testDebugUnitTest :core:test

      # 签名库由 app/build.gradle 直接指向仓库里的 keystore/<中文名>.jks，
      # 因此这里不需要再往 $HOME/.android 复制一份。
      #
      # 一旦要改签名的取用方式，务必记住：**必须**用仓库里那把固定的 keystore，
      # 不能现场生成 —— keytool -genkeypair 每次都会随机产生一把全新的密钥，
      # 签出来的包与设备上已装版本的签名对不上，Android 会直接拒绝覆盖安装
      # （表现为「应用未安装」，且不会告诉你原因）。

      - name: Read version
        id: ver
        run: |
          MAJ=$(grep -E '^versionMajor=' version.properties | cut -d= -f2 | tr -d '\r\n ')
          MIN=$(grep -E '^versionMinor=' version.properties | cut -d= -f2 | tr -d '\r\n ')
          PAT=$(grep -E '^versionPatch=' version.properties | cut -d= -f2 | tr -d '\r\n ')
          echo "name=${MAJ}.${MIN}.${PAT}" >> "$GITHUB_OUTPUT"
          echo "版本号: ${MAJ}.${MIN}.${PAT}"

      # v* 标签触发的构建必须与 version.properties 同号：标签写 5.2.1、包却是 5.2.0，
      # 是最难事后发现的一类发版错误（下载页、产物仓库、应用内更新全都对不上）。
      - name: Verify tag matches version
        if: ${{ github.ref_type == 'tag' }}
        run: node tools/check-version-tag.mjs

      # 更新说明取自版本日志（AppVersion.kt 里当前版本那一条），不另立一份发布说明 ——
      # 两份一定会对不上。应用检查到新版本时，会从蒲公英公开页把这段读回来：
      # 用户手里装的是旧版本，旧包的日志里当然没有新版本改了什么。
      # 写成文件而不是塞进环境变量：说明里有标点、引号，走文件由 curl 直接读，最稳。
      - name: Extract release notes
        run: |
          node tools/version-notes.js > "$RUNNER_TEMP/notes.txt"
          wc -c "$RUNNER_TEMP/notes.txt"

      - name: Build release APK
        run: ./gradlew --no-daemon assembleRelease

      # 权限断言（真实事故后的防线）：
      # INTERNET 一直是由某个第三方库通过「清单合并」带进安装包的，源码里从来没写过。
      # 3.4.2 做安装包瘦身删掉那批库时，它跟着一起消失，编译期毫无提示 ——
      # 结果整个应用的联网功能全废（账号、同步、反馈、更新），用户侧报「网络连接失败」。
      # 所以这里把底线写死：必须在，且必须没有已经移除的 CAMERA。断言失败就拦住这次出包。
      - name: Verify APK permissions
        run: |
          SDK="${ANDROID_HOME:-${ANDROID_SDK_ROOT:-/usr/local/lib/android/sdk}}"
          AAPT2=$(ls "$SDK"/build-tools/*/aapt2 2>/dev/null | sort -V | tail -1)
          if [ -z "$AAPT2" ]; then
            echo "::error::找不到 aapt2，无法做权限断言"
            exit 1
          fi
          APK=app/build/outputs/apk/release/app-release.apk
          "$AAPT2" dump badging "$APK" | grep -E "^uses-permission" || true
          "$AAPT2" dump badging "$APK" | grep -q "android.permission.INTERNET" \
            || { echo "::error::安装包缺少 INTERNET 权限 —— 所有联网功能都会失败，不要发布"; exit 1; }
          if "$AAPT2" dump badging "$APK" | grep -q "android.permission.CAMERA"; then
            echo "::error::安装包又带上了 CAMERA 权限（扫码早已移除，不该再有）"
            exit 1
          fi
          echo "权限断言通过：INTERNET 在，CAMERA 不在"

      # 对外分发的包只有这一个来源：本 job 当场构建的 app-release.apk。
      # 打包只是把它改成带版本号的文件名，不做任何二次加工（重签 / zipalign / 替换内容），
      # 因此后续上传、校验、下载页元数据全部指向同一份二进制。
      # 尺寸与 SHA-256 在这里算出并作为 job 输出传给 Publish：
      # 跨 job 靠附件传输文件，Publish 会再比一次哈希，确认拿回来的就是本次构建的包。
      - name: Package artifact
        id: package
        run: |
          APK=app/build/outputs/apk/release/app-release.apk
          OUT="CourseMeow-${{ steps.ver.outputs.name }}.apk"
          test -f "$APK"
          cp "$APK" "$OUT"
          SHA=$(sha256sum "$OUT" | awk '{print $1}')
          SIZE=$(stat -c %s "$OUT")
          echo "sha256=${SHA}" >> "$GITHUB_OUTPUT"
          echo "size=${SIZE}" >> "$GITHUB_OUTPUT"
          ls -lh "$OUT"

      # 下载页元数据（正式通道）：版本 / 体积 / SHA-256 / 直链全部由本次构建产物算出，
      # 与随后上传的 APK 是同一个文件。站点运行时优先读这份 /download/download-meta.json，
      # 静态兜底值只在下一次发版前有效 —— 这样「改页面」不再是发版必经的人工步骤。
      - name: Generate download metadata (release channel)
        if: ${{ inputs.channel == 'release' }}
        run: |
          set -euo pipefail
          VER="${{ steps.ver.outputs.name }}"
          APK="CourseMeow-${VER}.apk"
          test -s "$APK" || { echo "::error::找不到 $APK"; exit 1; }
          SIZE=$(stat -c %s "$APK")
          SHA=$(sha256sum "$APK" | awk '{print $1}')
          SIZE_LABEL="$(awk -v n="$SIZE" 'BEGIN{printf "%.1f", n/1048576}')MB"
          REL_DATE=$(TZ=Asia/Shanghai date +%Y.%m.%d)
          BUILT_AT=$(date -u +%Y-%m-%dT%H:%M:%SZ)
          jq -n \
            --arg version "$VER" \
            --argjson size "$SIZE" \
            --arg sizeLabel "$SIZE_LABEL" \
            --arg sha256 "$SHA" \
            --arg file "$APK" \
            --arg url "https://www.nexoraism.com/download/${APK}" \
            --arg releaseDate "$REL_DATE" \
            --arg builtAt "$BUILT_AT" \
            --arg source "github-actions/build-apk.yml" \
            '{version:$version, size:$size, sizeLabel:$sizeLabel, sha256:$sha256,
              file:$file, url:$url, releaseDate:$releaseDate, builtAt:$builtAt, source:$source}' \
            > download-meta.json
          cat download-meta.json

      # 调试版通道出的是 **perf 包**（`debuggable = false`，见 app/build.gradle 的 perf 构建类型）。
      #
      # 为什么不再出可调试包：可调试包在 ART 里几乎全靠解释 / JIT 执行，
      # 新页面首次绘制要 40~150ms，测试机上一路都在"卡"，但同一份代码的
      # 不可调试包只要十几毫秒（实测 120FPS / 掉帧 0.6%）。调试版本来就是给
      # 人试功能手感的，出的包必须接近正式版。
      #
      # perf 继承了 debug 的 applicationIdSuffix ".debug"、同一把签名与自更新地址，
      # 所以蒲公英仍然把它认成「调试版」那个独立应用，装到手机上也能覆盖旧调试包、保留数据。
      # 需要断点调试时，本地 `assembleDebug` 装自己机器即可，不走这条通道。
      - name: Build debug APK (debug channel)
        if: ${{ inputs.channel == '' || inputs.channel == 'debug' }}
        run: |
          ./gradlew --no-daemon assemblePerf
          test -f app/build/outputs/apk/perf/app-perf.apk
          cp app/build/outputs/apk/perf/app-perf.apk ./app-perf.apk
          ls -lh ./app-perf.apk

      # 产物交接：Publish job 拿到的必须是本次 run 里 build 刚刚产出的文件。
      # 附件保留 30 天，够覆盖「想补发时早已下班」的情况。
      - name: Prepare release notes for handoff
        run: cp "$RUNNER_TEMP/notes.txt" ./release-notes.txt

      # overwrite: 整条 run「全部重跑」时会重新上传同名附件，v4 默认遇到重名会直接报错，
      # 这里显式允许覆盖，保证重跑可用。
      - name: Upload release APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: release-apk
          path: CourseMeow-${{ steps.ver.outputs.name }}.apk
          retention-days: 30
          overwrite: true

      - name: Upload release notes artifact
        uses: actions/upload-artifact@v4
        with:
          name: build-notes
          path: release-notes.txt
          retention-days: 30
          overwrite: true

      - name: Upload download metadata artifact (release channel)
        if: ${{ inputs.channel == 'release' }}
        uses: actions/upload-artifact@v4
        with:
          name: download-meta
          path: download-meta.json
          retention-days: 30
          overwrite: true

      - name: Upload perf APK artifact (debug channel)
        if: ${{ inputs.channel == '' || inputs.channel == 'debug' }}
        uses: actions/upload-artifact@v4
        with:
          name: perf-apk
          path: app-perf.apk
          retention-days: 30
          overwrite: true

      - name: Build summary
        if: success()
        run: |
          {
            echo "### 构建成功（未对外发布）"
            echo
            echo "- 版本：${{ steps.ver.outputs.name }}"
            echo "- 通道：${{ inputs.channel || 'debug' }}"
            echo "- APK 字节数：${{ steps.package.outputs.size }}"
            echo "- SHA-256：\`${{ steps.package.outputs.sha256 }}\`"
            echo "- 产物仓库 / 自托管 / 蒲公英由独立 job「Publish」执行：它失败只会计到 Publish，不影响本次构建结论。"
          } >> "$GITHUB_STEP_SUMMARY"

  # ---------------------------------------------------------------------------- 发布
  # 只做对外分发，不重新构建。失败时不会把上面的 build 判红；
  # Actions 里「Re-run failed jobs」只会重跑本 job，附件里的包直接复用。
  publish:
    name: Publish (产物仓库 / 自托管 / 蒲公英)
    needs: build
    if: ${{ needs.build.result == 'success' }}
    runs-on: ubuntu-latest
    # 每一步都有单次硬上限、重试轮数也有上限：最坏情况（全部步骤都反复失败）约 39 分钟，
    # 给 45 分钟兜底。健康网络下整段只有几十秒 —— 不会再有「一步卡住 36 分钟」。
    timeout-minutes: 45

    # 只有本 job 需要写仓库：gh release 覆盖 Android-Latest（删旧附件、传新包）。
    # 工作流级是 contents: read，这一行把写权限收在这里，构建 job 拿不到。
    permissions:
      contents: write

    env:
      # 蒲公英 API Key。注意：key 是**账号级唯一**的，只用于认证账号；
      # 「上传到哪个应用」由安装包本身决定（蒲公英按包名/签名识别归属）。
      # 因此不用也不能按应用配多把 key —— debug 包上传会进调试应用、release 包进正式应用。
      # 没配就自动跳过上传那一步，不影响出包。
      PGYER_API_KEY: ${{ secrets.PGYER_API_KEY }}
      # 安卓端产物仓库的固定标签
      ANDROID_RELEASE: Android-Latest

    steps:
      - name: Download release APK
        uses: actions/download-artifact@v4
        with:
          name: release-apk

      - name: Download release notes
        uses: actions/download-artifact@v4
        with:
          name: build-notes

      - name: Download perf APK (debug channel)
        if: ${{ inputs.channel == '' || inputs.channel == 'debug' }}
        uses: actions/download-artifact@v4
        with:
          name: perf-apk

      - name: Download download metadata (release channel)
        if: ${{ inputs.channel == 'release' }}
        uses: actions/download-artifact@v4
        with:
          name: download-meta

      # 交接校验：附件里拿回来的包必须和 build job 当场算出的 SHA-256 / 字节数一致。
      # 这一步替代原来「同目录下 app-release.apk 与打包件比哈希」的检查 ——
      # 跨 job 后唯一可信的锚点就是 build 输出的哈希值。
      - name: Verify handoff matches build output
        run: |
          set -euo pipefail
          VER="${{ needs.build.outputs.version }}"
          APK="CourseMeow-${VER}.apk"
          test -s "$APK" || { echo "::error::附件里没有 $APK"; exit 1; }
          GOT_SHA=$(sha256sum "$APK" | awk '{print $1}')
          GOT_SIZE=$(stat -c %s "$APK")
          echo "附件：${GOT_SIZE} 字节  SHA-256=${GOT_SHA}"
          [ "$GOT_SHA" = "${{ needs.build.outputs.sha256 }}" ] \
            || { echo "::error::附件 SHA-256 与本次构建不一致（$GOT_SHA vs ${{ needs.build.outputs.sha256 }}）"; exit 1; }
          [ "$GOT_SIZE" = "${{ needs.build.outputs.size }}" ] \
            || { echo "::error::附件字节数与本次构建不一致（$GOT_SIZE vs ${{ needs.build.outputs.size }}）"; exit 1; }

      # 把这一版的安装包放进产物仓库：固定标签 Android-Latest 的发布，先清掉上一版附件再传新的，
      # 所以那里始终只有一份最新安装包。本机 `scripts/发布蒲公英.bat` 就是从这里取包。
      - name: Publish to artifact release
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          # 本 job 没有 checkout，gh 找不到 git remote，必须显式指定仓库
          GH_REPO: ${{ github.repository }}
        run: |
          set -euo pipefail
          VER="${{ needs.build.outputs.version }}"
          APK="CourseMeow-${VER}.apk"
          NOTES_FILE=release-notes.txt
          test -s "$NOTES_FILE" || echo "CourseMeow ${VER}" > "$NOTES_FILE"
          ok=0
          for attempt in 1 2 3; do
            echo "== 产物仓库发布 ${attempt}/3 =="
            if (
              set -euo pipefail
              if gh release view "$ANDROID_RELEASE" >/dev/null 2>&1; then
                # 只留最新一版：先把已有的附件逐个删掉
                for name in $(gh release view "$ANDROID_RELEASE" --json assets -q '.assets[].name'); do
                  gh release delete-asset "$ANDROID_RELEASE" "$name" -y >/dev/null || true
                done
                gh release edit "$ANDROID_RELEASE" --title "拾课喵 ${VER}" --notes-file "$NOTES_FILE"
              else
                gh release create "$ANDROID_RELEASE" --title "拾课喵 ${VER}" --notes-file "$NOTES_FILE"
              fi
              gh release upload "$ANDROID_RELEASE" "$APK" --clobber
            ); then
              ok=1
              break
            fi
            echo "::warning::产物仓库发布第 ${attempt} 次失败，15s 后重试"
            sleep 15
          done
          [ "$ok" = "1" ] || { echo "::error::产物仓库（$ANDROID_RELEASE）发布连续 3 次失败"; exit 1; }
          {
            echo "### 产物仓库（永远只留最新一版）"
            echo
            echo "- 版本：${VER}"
            echo "- 附件：${APK}"
            echo "- 发布页：https://github.com/Syntax-Liu/Course-Meow/releases/tag/${ANDROID_RELEASE}"
          } >> "$GITHUB_STEP_SUMMARY"

      # 正式版自托管直链（必做）：下载页的版本 / 体积 / SHA-256 / 直链由产物元数据驱动，
      # 不传这一步，下个正式版一发布主下载入口就是 404。
      # 输入**只接受本次 build job 的产物**（上一步已校验哈希）：
      # 工作流本身没有「传入本地 APK」的入口，附件也只来自本次 run。
      # 顺序：上传 APK + 元数据 → 匿名校验（200 / content-length / content-type / SHA-256）
      #       → 元数据一致 → 清理更早版本。校验一步都不省。
      # 凭据复用 deploy-web.yml 的 CLOUDBASE_SECRET_ID / CLOUDBASE_SECRET_KEY；
      # 没配就直接失败，不能静默出一版「站点指向不存在的文件」的正式版。
      #
      # 网络兜底（2026-11-08 事故复盘）：CloudBase CLI 自己会串行重试很久，一步能拖 36 分钟。
      # 这里给每一次调用套 `timeout`，失败整体重来一轮，最多 3 轮；并给出明确的重试日志。
      - name: Upload self-hosted APK (release channel)
        if: ${{ inputs.channel == 'release' }}
        env:
          TCB_ENV: coursemeow-d6gywwh2kd8dca6c8
          SITE_BASE: https://www.nexoraism.com
          TCB_SECRET_ID: ${{ secrets.CLOUDBASE_SECRET_ID }}
          TCB_SECRET_KEY: ${{ secrets.CLOUDBASE_SECRET_KEY }}
        run: |
          set -euo pipefail
          VER="${{ needs.build.outputs.version }}"
          APK="CourseMeow-${VER}.apk"
          BUILT_SHA="${{ needs.build.outputs.sha256 }}"
          BUILT_SIZE="${{ needs.build.outputs.size }}"
          META="download-meta.json"
          test -s "$APK" || { echo "::error::找不到 $APK"; exit 1; }
          test -s "$META" || { echo "::error::缺少下载元数据 $META"; exit 1; }
          if [ -z "${TCB_SECRET_ID}" ] || [ -z "${TCB_SECRET_KEY}" ]; then
            echo "::error::缺少 CLOUDBASE_SECRET_ID / CLOUDBASE_SECRET_KEY，无法上传自托管直链"
            exit 1
          fi
          LOCAL_SHA=$(sha256sum "$APK" | awk '{print $1}')
          LOCAL_SIZE=$(stat -c %s "$APK")
          if [ "$LOCAL_SHA" != "$BUILT_SHA" ] || [ "$LOCAL_SIZE" != "$BUILT_SIZE" ]; then
            echo "::error::待上传的 $APK 与本次构建产物不一致（$LOCAL_SHA / $LOCAL_SIZE）"
            exit 1
          fi
          # 元数据也必须是本次构建生成的那一份（版本 / 体积 / SHA 与包对得上）
          jq -e --arg v "$VER" --argjson n "$LOCAL_SIZE" --arg s "$LOCAL_SHA" \
            '.version==$v and .size==$n and .sha256==$s' "$META" >/dev/null \
            || { echo "::error::$META 与本次构建产物不一致"; exit 1; }
          echo "本地：${LOCAL_SIZE} 字节  SHA-256=${LOCAL_SHA}（与本次构建产物一致）"

          # 单次调用加硬超时 + 有界重试：任一次失败/超时都整轮重来，最多 max 轮。
          # limit 用秒。上传给 4 分钟：4 兆的包在 4 分钟内传不完（低于 20KB/s）基本可判定链路不可用，
          # 与其像事故里那样让 CLI 自己串行重试 36 分钟，不如尽快失败、整轮重来。
          retry() {
            label="$1"; max="$2"; wait_s="$3"; limit="$4"
            shift 4
            n=1
            while :; do
              echo "== ${label}：第 ${n}/${max} 次（单次上限 $((limit / 60)) 分钟）=="
              if timeout "$limit" "$@"; then
                return 0
              fi
              if [ "$n" -ge "$max" ]; then
                echo "::error::${label} 连续 ${max} 次失败或超时"
                return 1
              fi
              echo "::warning::${label} 第 ${n} 次失败或超时，${wait_s}s 后重试"
              n=$((n + 1))
              sleep "$wait_s"
            done
          }

          retry "安装 CloudBase CLI" 2 15 180 npm install -g @cloudbase/cli@3.9.1 --no-audit --no-fund
          retry "登录 CloudBase" 2 15 90 tcb login --apiKeyId "$TCB_SECRET_ID" --apiKey "$TCB_SECRET_KEY"
          # cloudPath 必须写到文件名，写目录会把对象名变成目录名
          retry "上传 APK" 3 20 240 tcb hosting deploy "$APK" "/download/${APK}" -e "$TCB_ENV" --json
          # 元数据与 APK 同目录、同一次上传；站点运行时读它来渲染下载信息
          retry "上传下载元数据" 3 20 90 tcb hosting deploy "$META" "/download/download-meta.json" -e "$TCB_ENV" --json

          URL="${SITE_BASE}/download/${APK}"
          ok=0
          for attempt in 1 2 3; do
            q="?v=$(date +%s)${attempt}"
            head=$(curl -sS -I --connect-timeout 20 --max-time 30 "${URL}${q}" | tr -d '\r' || true)
            code=$(printf '%s\n' "$head" | awk 'NR==1{print $2}')
            clen=$(printf '%s\n' "$head" | awk -F': ' 'tolower($1)=="content-length"{print $2}')
            ctype=$(printf '%s\n' "$head" | awk -F': ' 'tolower($1)=="content-type"{print $2}')
            dl="$RUNNER_TEMP/coursemeow-apk-check.apk"
            curl -sS -L --connect-timeout 20 --max-time 150 -o "$dl" "${URL}${q}" || true
            got_size=$(stat -c %s "$dl" 2>/dev/null || echo 0)
            got_sha=$(sha256sum "$dl" 2>/dev/null | awk '{print $1}' || echo "")
            echo "校验 ${attempt}/3：HTTP=${code} content-length=${clen} 下载=${got_size} type=${ctype} sha=${got_sha}"
            if [ "$code" = "200" ] && [ "$clen" = "$LOCAL_SIZE" ] && [ "$got_size" = "$LOCAL_SIZE" ] \
              && [ "$got_sha" = "$LOCAL_SHA" ] && printf '%s' "$ctype" | grep -qi 'application/vnd.android.package-archive'; then
              ok=1
              break
            fi
            sleep 15
          done
          [ "$ok" = "1" ] || { echo "::error::自托管直链校验失败：$URL（包已上传，但四项校验没过，先别对外宣告发版）"; exit 1; }

          # 元数据校验：线上 JSON 必须与本次产物一致，站点才会显示正确的版本 / 体积 / SHA / 直链。
          # 少了这一步，页面可能拿旧值去描述新包，比 404 更难发现。
          META_URL="${SITE_BASE}/download/download-meta.json"
          mok=0
          for attempt in 1 2 3; do
            body=$(curl -sS --connect-timeout 20 --max-time 30 "${META_URL}?v=$(date +%s)${attempt}" || true)
            if printf '%s' "$body" | jq -e \
              --arg v "$VER" --argjson n "$LOCAL_SIZE" --arg s "$LOCAL_SHA" --arg u "$URL" \
              '.version==$v and .size==$n and .sha256==$s and .url==$u' >/dev/null 2>&1; then
              mok=1
              echo "元数据校验 ${attempt}/3：一致"
              break
            fi
            echo "元数据校验 ${attempt}/3：暂时不一致（可能仍在 CDN 生效中）"
            printf '%s' "$body" | head -c 400 || true
            printf '\n'
            sleep 15
          done
          [ "$mok" = "1" ] || { echo "::error::线上 download-meta.json 与产物不一致：$META_URL"; exit 1; }

          # 保留最近 2 个版本（新版 + 上一版可回滚）；只删具体文件，不用 --dir，避免误删整个目录。
          # 清理是收尾动作：失败只记 warning，不让「已经校验通过的发布」因为删旧文件而变红。
          echo "== 清理更早的自托管版本（保留最近 2 个）=="
          LIST=$(timeout 120 tcb hosting list /download -e "$TCB_ENV" --json || true)
          echo "$LIST"
          mapfile -t FILES < <(printf '%s' "$LIST" | grep -oE 'CourseMeow-[0-9]+(\.[0-9]+){2}\.apk' | sort -u | sort -V)
          COUNT=${#FILES[@]}
          echo "远端共 ${COUNT} 个：${FILES[*]:-（无）}"
          if [ "$COUNT" -gt 2 ]; then
            i=0
            while [ "$i" -lt $((COUNT - 2)) ]; do
              OLD="${FILES[$i]}"
              if [ "$OLD" != "$APK" ]; then
                echo "删除旧版本：/download/$OLD"
                timeout 120 tcb hosting delete "/download/$OLD" -e "$TCB_ENV" --json \
                  || echo "::warning::删除 $OLD 失败，请手动清理"
              fi
              i=$((i + 1))
            done
          fi

          {
            echo "### 自托管直链（正式版）"
            echo
            echo "- 直链：${URL}"
            echo "- 体积：${LOCAL_SIZE} 字节"
            echo "- SHA-256：\`${LOCAL_SHA}\`"
            echo "- 元数据：${META_URL}（版本 / 体积 / SHA-256 / 直链与产物一致）"
            echo "- 校验：HTTP 200 / content-length / content-type / SHA-256 全部一致"
          } >> "$GITHUB_STEP_SUMMARY"

      # 上传到蒲公英（内测分发），给测试用户一个「一个链接 + 一张二维码，永远指向最新版」的入口。
      #
      # 为什么需要它：腾讯的 QQ / 微信在转发 apk 时会把后缀改成 .apk.1（有意为之，为的是提高
      # 恶意软件传播成本），部分系统收到这种文件连安装包都认不出来，测试用户每次都得手动重命名；
      # 而仓库是私有的，GitHub 的下载地址测试用户也打不开。蒲公英的下载页和二维码**跨版本不变**，
      # 群里贴一次就行，不用每出一个版本换一次链接，也没有改后缀这回事。
      #
      # 用的是旧版单次上传接口：一次请求就把文件传完并返回下载页信息，CI 里最好维护。
      # 文件大了以后若嫌慢，可以换成官方推荐的 getCOSToken → 直传 COS → buildInfo 三步流程。
      # 失败只记 warning（分发失败不该让发版变红：安装包本身没问题，用户仍可从产物仓库取）。
      - name: Upload to Pgyer
        # 两条流程的区别就在这一行：
        #   upload_pgyer=true  → 出包后上传蒲公英；
        #   upload_pgyer=false → 只进产物仓库，不上蒲公英（给测试用户 GitHub 取包）。
        # 上传到哪个蒲公英应用由 channel 决定（debug/release 各一把 key），脚本里选。
        # workflow_dispatch 手动触发时 inputs 可能为空，这里兜底默认为 true。
        if: ${{ inputs.upload_pgyer == '' || fromJSON(inputs.upload_pgyer) }}
        env:
          CHANNEL: ${{ inputs.channel || 'debug' }}
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          set -euo pipefail
          VER="${{ needs.build.outputs.version }}"
          # 按通道选「包」，key 用账号级唯一的 PGYER_API_KEY：
          #   正式版用 release 包（com.nexoraism.coursemeow）      → 蒲公英进正式应用；
          #   调试版用 perf 包（com.nexoraism.coursemeow.debug）    → 蒲公英进调试应用。
          # 两个包包名不同 → 两个应用 → 两条互不覆盖的下载链接。
          # perf 与 release 都不可调试；区别只是 release 额外开了 R8。
          KEY="${PGYER_API_KEY}"
          if [ "$CHANNEL" = "release" ]; then
            APK="CourseMeow-${VER}.apk"
            CHANNEL_NAME="正式版"
          else
            # 调试版通道传的是 perf 包（不可调试、但包名仍是 .debug）
            APK=app-perf.apk
            CHANNEL_NAME="调试版"
          fi
          test -f "$APK" || { echo "::error::找不到 $APK"; exit 1; }
          if [ -z "$KEY" ]; then
            echo "::warning::未配置蒲公英 key（PGYER_API_KEY），跳过上传，安装包已在产物仓库"
            exit 0
          fi
          set +e
          # 更新说明走文件（`-F name=<file`）：直接从 AppVersion.kt 抽出来的那一行，
          # 里面有中文标点与引号，塞进命令行容易被 curl 的 -F 语法（; 与 , 有特殊含义）曲解。
          NOTES_FILE=release-notes.txt
          if [ ! -s "$NOTES_FILE" ]; then
            echo "CourseMeow ${VER}" > "$NOTES_FILE"
          fi
          # 带上连接/整体超时：原来没设上限，网络抽风时这一步同样能把 job 拖死。
          RESP=$(curl -sS --connect-timeout 30 --max-time 600 \
            -F "_api_key=${KEY}" \
            -F "file=@${APK}" \
            -F "buildUpdateDescription=<${NOTES_FILE}" \
            https://upload.pgyer.com/apiv2/app/upload)
          set -e
          echo "$RESP"
          # 响应为空或不是 JSON 时 jq 会非零退出，这里兜住，别让它把整步弄红
          CODE=$(echo "$RESP" | jq -r '.code // 999' 2>/dev/null) || CODE=999
          if [ "$CODE" != "0" ]; then
            # 分发失败不该让构建变红：安装包本身没问题，用户仍可从产物仓库取。
            echo "::warning::蒲公英上传失败（code=$CODE）：$(echo "$RESP" | jq -r '.message')"
            exit 0
          fi
          SHORT=$(echo "$RESP" | jq -r '.data.buildShortcutUrl // .buildShortcutUrl // empty')
          QR=$(echo "$RESP" | jq -r '.data.buildQRCodeURL // .buildQRCodeURL // empty')
          {
            echo "### 蒲公英下载页（$CHANNEL_NAME · 跨版本固定，贴一次就行）"
            echo
            echo "- 链接：https://www.pgyer.com/${SHORT}"
            echo "- 二维码：${QR}"
          } >> "$GITHUB_STEP_SUMMARY"

      # 收尾状态：不管发布成功还是失败都写一行，方便一眼区分「构建成功、发布失败」。
      - name: Publish summary
        if: always()
        run: |
          {
            echo "### 发布结果：${{ job.status }}"
            echo
            echo "- 构建（build job）：${{ needs.build.result }}（安装包已产出，SHA-256 \`${{ needs.build.outputs.sha256 }}\`）"
            echo "- 版本：${{ needs.build.outputs.version }}"
            if [ "${{ job.status }}" != "success" ]; then
              echo "- 失败后处理：在 Actions 本 run 上点「Re-run failed jobs」只重跑本 job，不会重新构建；"
              echo "  附件在本次 run 内保留 30 天，可直接复用。"
            fi
          } >> "$GITHUB_STEP_SUMMARY"
