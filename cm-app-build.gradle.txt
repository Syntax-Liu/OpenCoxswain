// KSP 会把生成物同时放在 generated/ksp/<variant>/java 和 .../java/byRounds/<n> 两处，
// 干净构建（没有任何增量缓存）时这两份都会被 javac 收进去，报「重复类」而失败。
// byRounds 那份是 KSP 给自己做增量用的中间副本，编译时排除掉即可。
tasks.withType(JavaCompile).configureEach { t ->
    t.exclude '**/byRounds/**'
}

ext {
    vpFile = rootProject.file("version.properties")
    vMajor = 0
    vMinor = 0
    vPatch = 0
    if (vpFile.exists()) {
        vpFile.readLines().each { line ->
            def trimmed = line.trim()
            if (trimmed.startsWith("versionMajor=")) {
                vMajor = trimmed.split("=", 2)[1].trim().toInteger()
            } else if (trimmed.startsWith("versionMinor=")) {
                vMinor = trimmed.split("=", 2)[1].trim().toInteger()
            } else if (trimmed.startsWith("versionPatch=")) {
                vPatch = trimmed.split("=", 2)[1].trim().toInteger()
            }
        }
    }
    computedVersionName = "${vMajor}.${vMinor}.${vPatch}"
    computedVersionCode = vMajor * 10000 + vMinor * 100 + vPatch
}

// 以后加字段能一眼看出哪一版改了表，迁移漂移不再是「凭记忆」。

android {
    // namespace 保持不变：它只是源码包名与 R 类的归属，装在手机上的「包名」由
    // 下面的 applicationId 决定。改 namespace 要动几百个文件的 package 声明，
    // 对使用者没有任何可见影响，所以不动。
    namespace "com.nexoraism.coursemeow"
    compileSdk 34

    defaultConfig {
        applicationId "com.nexoraism.coursemeow"
        minSdk 26
        targetSdk 34
        versionCode computedVersionCode
        versionName computedVersionName

        // 只带真机用得上的两种架构。
        //
        // 原生库现在只剩 androidx 自带的那几个（graphics-path、datastore 计数器，各几 KB），
        // 留着这个过滤是为了「不管将来哪个依赖带进 .so，都不会顺手把模拟器专用的
        // x86 / x86_64 也打进包里」。arm64 覆盖现在所有手机，armeabi-v7a 兜住老机器。
        // 代价：x86 模拟器上装不了（本项目一律用真机验证）。
        ndk {
            abiFilters "arm64-v8a", "armeabi-v7a"
        }

        // 仪器化测试（app/src/androidTest）跑在设备/模拟器上，数据库迁移回放依赖它。
        testInstrumentationRunner "androidx.test.runner.AndroidJUnitRunner"
    }

    sourceSets {
        // Room 的 MigrationTestHelper 从 assets 里读历史 schema（app/schemas/*.json）。
        // 不把 schema 目录挂进 androidTest 的 assets，createDatabase(旧版本) 就找不到基线。
        androidTest.assets.srcDirs += files("$projectDir/schemas")
    }

    signingConfigs {
        debug {
            // 签名库固定在仓库里（keystore/桃花坞里柳边风.jks），本地与 CI 用同一把，
            // 因此两边出的包可以互相覆盖安装。
            //
            // 千万别改成「各机器自己生成一把」：keytool -genkeypair 每次都是全新的密钥，
            // 与设备上已装版本的签名对不上，Android 会直接拒绝覆盖安装
            // （表现为「应用未安装」，而且不告诉你原因）。
            //
            // 文件名与别名都是中文/大小写敏感的，改这里之前先 keytool -list 看一下实际值。
            storeFile rootProject.file("keystore/桃花坞里柳边风.jks")
            // 口令与别名**不写在这里**（体检 E-01，用户决定按方案 A 落地）：
            // 取值优先级 = 项目 / 本机 gradle.properties → 环境变量
            // ORG_GRADLE_PROJECT_coursemeowStorePassword 等 → 手动读 ~/.gradle/gradle.properties。
            // 最后一条是给「用 -g 指定了独立 Gradle Home」的场景兜底：那种情况下 Gradle
            // 不会读用户目录的 gradle.properties，本机 tools/发调试版快.ps1 就是这种用法。
            // 千万别现场生成一把新 key 顶替 —— 签名一变，已发布包就无法覆盖升级。
            def readSigningValue = { String key, String fallback ->
                def fromGradle = providers.gradleProperty(key).getOrElse("").trim()
                if (fromGradle && fromGradle != "CHANGE_ME") return fromGradle
                def homeProps = new File(System.getProperty("user.home"), ".gradle/gradle.properties")
                if (homeProps.isFile()) {
                    def props = new Properties()
                    homeProps.withInputStream { input -> props.load(input) }
                    def fromHome = (props.getProperty(key) ?: "").trim()
                    if (fromHome && fromHome != "CHANGE_ME") return fromHome
                }
                return fallback
            }
            def storePwd = readSigningValue("coursemeowStorePassword", "")
            def keyPwd = readSigningValue("coursemeowKeyPassword", "")
            if (storePwd.isEmpty() || keyPwd.isEmpty()) {
                throw new GradleException(
                    "Missing signing password: set coursemeowStorePassword / coursemeowKeyPassword " +
                        "in ~/.gradle/gradle.properties (or ORG_GRADLE_PROJECT_* env / repo Secrets). " +
                        "Do not edit the repository's gradle.properties."
                )
            }
            storePassword storePwd
            keyAlias readSigningValue("coursemeowKeyAlias", "CourseMeow")
            keyPassword keyPwd

            // v1 / v2 / v3 三种签名方案全开。
            //
            // minSdk 是 26，AGP 会因此默认只签 v2 —— 对系统安装器够用，
            // 但对第三方安装器与兼容环境不够：它们判断「这个包签没签名」的常见做法
            // 是去 META-INF 里找 *.SF / *.RSA，而 v2 签名是把签名块塞在 ZIP 中央目录之后，
            // 这些包里一个签名文件都没有，于是被判成「未签名包」直接拒绝安装
            // （例：鸿蒙上通过卓易通跑安卓应用的那类容器）。
            // 多带一层 v1 只是让安装包大一点点，换的是这类环境肯认它。
            enableV1Signing = true
            enableV2Signing = true
            enableV3Signing = true
        }
    }

    buildTypes {
        debug {
            // 给调试包一个独立的 applicationId 后缀。
            //
            // 蒲公英按「包名」识别应用归属，不看签名。此前 debug 和 release 的包名
            // 都是 com.nexoraism.coursemeow、签名也相同，两个包被蒲公英认成同一个应用：
            // 调试版一传上去要么覆盖正式版、要么被拒，永远等不到一条独立的调试链接。
            // 加这个后缀后定包名变成 com.nexoraism.coursemeow.debug，蒲公英才会当成
            // 独立的第二个应用，给出和正式版互不覆盖的调试版下载链接；
            // 手机上两个包也因包名不同而能共存安装。
            //
            // 权限/组件适配：FileProvider 的 authority 写成 ${applicationId}.fileprovider，
            // 会随后缀自动变成 com.nexoraism.coursemeow.debug.fileprovider，无冲突。
            //
            // 签名沿用仓库固定 keystore（与 release 同一把）——包名已经隔开，
            // 不需要也不应该再换签名，避免引入「各机器各生成一把」的老坑。
            applicationIdSuffix ".debug"
            signingConfig signingConfigs.debug
            // 模拟器是 x86_64 的，而 defaultConfig 只打 arm 两种架构，装上去会报
            // 「no matching ABI」。调试变体多带一份 x86_64（原生库只有 androidx 那几个
            // 几十 KB 的 .so），正式包仍然保持 arm-only。
            ndk {
                abiFilters "arm64-v8a", "armeabi-v7a", "x86_64"
            }
            // 调试版的自更新指向调试版蒲公英主页（独立短链）。
            // 不写死：否则调试包和正式包都会去读正式主页，两版版本号一致时就
            // 判成「没有新版本」（isNewer 走 40009 > 40009 为假），应用内根本下不了。
            buildConfigField "String", "PGYER_PAGE_URL", "\"https://www.pgyer.com/CourseMeow-Android-debug\""
        }

        release {
            // 打开 R8 与资源压缩。
            //
            // 之前这里是 false，代价很实在：material-icons-extended 里一万多个图标
            // 一个不落地全进了包，而应用实际只用了几十个 —— 光这一项就让安装包
            // 到了 26 MB（dex 解压后 51 MB，其中 classes.dex 一项就 29 MB）。
            //
            // 混淆刻意**不开**：崩溃日志是排查线上问题的唯一手段，堆栈里必须留着
            // 类名与方法名（`TimeCalc.kt:61`），混淆后就变成 `a.b.c` 了。
            // 只做「删掉没人用的代码」，不改名字 —— 见 proguard-rules.pro。
            minifyEnabled true
            shrinkResources true
            signingConfig signingConfigs.debug
            // 正式版自更新指向正式版蒲公英主页（默认短链）。
            buildConfigField "String", "PGYER_PAGE_URL", "\"https://www.pgyer.com/CourseMeow-Android\""
            proguardFiles(
                getDefaultProguardFile('proguard-android-optimize.txt'),
                "proguard-rules.pro"
            )
        }

        /**
         * 调试版通道实际出这个包：包名、签名、自更新地址与 debug 完全一致（装上去直接
         * 覆盖、保留数据），但 **debuggable = false**、也不开 R8。
         *
         * 原因：可调试包在 ART 里几乎全靠解释/JIT，首次绘制一个元素要 0.3~0.6ms，
         * 不可调试包只要 0.05ms —— 同样代码能差 3~5 倍帧耗时，用户装调试版会觉得卡。
         * 断点调试请仍用 assembleDebug（app-debug.apk），不要走这条通道。
         */
        perf {
            initWith debug
            debuggable false
            minifyEnabled false
        }
    }

tasks.register("bumpVersion") {
    doLast {
        def kind = project.findProperty("kind")?.toString()?.toLowerCase() ?: "patch"
        def nextMajor = vMajor
        def nextMinor = vMinor
        def nextPatch = vPatch
        if (kind == "major") {
            nextMajor = vMajor + 1
            nextMinor = 0
            nextPatch = 0
        } else if (kind == "minor") {
            nextMinor = vMinor + 1
            nextPatch = 0
        } else {
            nextPatch = vPatch + 1
        }
        vpFile.text = "versionMajor=" + nextMajor + "\nversionMinor=" + nextMinor + "\nversionPatch=" + nextPatch + "\n"
        def nextCode = nextMajor * 10000 + nextMinor * 100 + nextPatch
        logger.lifecycle("Version bumped: ${computedVersionName} (${computedVersionCode}) -> ${nextMajor}.${nextMinor}.${nextPatch} (${nextCode})")
    }
}

tasks.register("copyAudit") {
    group = "verification"
    description = "Check UI text copy consistency"

    doLast {
        def script = rootProject.file("tools/copy-audit.js")
        if (!script.exists()) {
            logger.lifecycle("Copy audit: skipped (tools/copy-audit.js not found)")
            return
        }
        try {
            def proc = new ProcessBuilder("node", script.absolutePath, "--quiet")
                .directory(rootProject.projectDir)
                .redirectErrorStream(true)
                .start()
            def output = new String(proc.inputStream.readAllBytes(), "UTF-8").trim()
            def code = proc.waitFor()
            if (code == 0) {
                logger.lifecycle("Copy audit: all clean.")
            } else {
                def report = rootProject.file("build/copy-audit-report.txt")
                report.parentFile.mkdirs()
                report.text = output
                logger.lifecycle("Copy audit: issues found, see build/copy-audit-report.txt")
            }
        } catch (Exception e) {
            logger.lifecycle("Copy audit: skipped (node not available: ${e.message})")
        }
    }
}

// A 类适配器回放：在 Node 里把内置脚本跑一遍，校验桥交互顺序、课程/配置/作息/组合作息落库内容。
// 与 copyAudit 一样挂在 preBuild 上，本机没有 node 时只提示跳过，不阻断构建。
tasks.register("replayAdapters") {
    group = "verification"
    description = "Replay built-in A-class adapters against recorded fixtures"

    doLast {
        def script = rootProject.file("tools/replay-adapters.mjs")
        if (!script.exists()) {
            logger.lifecycle("Adapter replay: skipped (tools/replay-adapters.mjs not found)")
            return
        }
        try {
            def proc = new ProcessBuilder("node", script.absolutePath)
                .directory(rootProject.projectDir)
                .redirectErrorStream(true)
                .start()
            def output = new String(proc.inputStream.readAllBytes(), "UTF-8").trim()
            def code = proc.waitFor()
            if (code == 0) {
                logger.lifecycle("Adapter replay: ${output.replace('\n', ' | ')}")
            } else {
                def report = rootProject.file("build/adapter-replay-report.txt")
                report.parentFile.mkdirs()
                report.text = output
                logger.lifecycle("Adapter replay: FAILED, see build/adapter-replay-report.txt")
                throw new GradleException("适配器回放未通过，详见 build/adapter-replay-report.txt")
            }
        } catch (GradleException e) {
            throw e
        } catch (Exception e) {
            logger.lifecycle("Adapter replay: skipped (node not available: ${e.message})")
        }
    }
}
